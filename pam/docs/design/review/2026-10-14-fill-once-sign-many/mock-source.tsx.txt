function Progress({ pct, label }: { readonly pct: number; readonly label: string }) {
  return (
    <VStack gap={2} xstyle={styles.row}>
      <HStack justify="between" xstyle={styles.row}>
        <Text type="supporting">{label}</Text>
        <Text type="supporting">{pct}%</Text>
      </HStack>
      <HStack xstyle={styles.track}>
        <HStack xstyle={styles.fill(pct)} />
      </HStack>
    </VStack>
  );
}

/** Every note is Astryx's `Banner`: a confirmation (something is done) is `success`, green (Will, 06:28: "good news in green"); an informational note stays the subtle blue `info` one the place page uses. */
let CONFIRM_AS: 'info' | 'success' = 'success';
function Calm({ title, body, confirm }: { readonly icon?: unknown; readonly confirm?: boolean; readonly title?: string; readonly body: string }) {
  return <Banner status={confirm ? CONFIRM_AS : 'info'} title={title ?? body} description={title ? body : undefined} />;
}

/** One line of a form as it sits on the paper: the field's name small, its filled-in value under it. */
function PaperField({ label, value, wide }: { readonly label: string; readonly value: string; readonly wide?: boolean }) {
  return (
    <VStack gap={0} xstyle={wide ? styles.row : styles.grow}>
      <Text type="supporting" xstyle={styles.paperQuiet}>
        {label}
      </Text>
      <HStack xstyle={styles.paperField}>
        <Text xstyle={styles.paperText}>{value}</Text>
      </HStack>
    </VStack>
  );
}

function BackBar({ title, sub, pct }: { readonly title: string; readonly sub: string; readonly pct: number }) {

function Page() {
  const raw = useSearchParams().get('s') ?? 'fill';
  return <Screen which={raw.replace('-success', '')} />;
}

export default function DesignSign() {
  return (

    </Suspense>
  );
}
